## 3. 模块设计

### 3.1 主进程模块

#### 3.1.1 SerialManager (串口管理器)

```javascript
/**
 * 串口管理器 - 负责所有串口操作
 */
class SerialManager extends EventEmitter {
  // 属性
  ports: Map<string, SerialPort>  // 已打开的串口

  // 方法
  listPorts(): Promise<PortInfo[]>           // 列出可用串口
  open(config: SerialConfig): Promise<void>  // 打开串口
  close(portPath: string): Promise<void>     // 关闭串口
  write(portPath: string, data: Buffer): Promise<void>  // 写入数据

  // 事件
  on('data', (portPath, data) => {})    // 接收数据事件
  on('error', (portPath, error) => {})  // 错误事件
  on('close', (portPath) => {})         // 关闭事件
}
```

#### 3.1.2 IPCHandler (IPC 处理器)

```javascript
/**
 * IPC 通信处理 - 主进程端

const IPCChannels = {
```

### 3.3 Preload 脚本

```javascript
// preload.js - 安全地暴露 API 给渲染进程
const { contextBridge, ipcRenderer } = require('electron');

contextBridge.exposeInMainWorld('electronAPI', {
  // 串口操作
  listPorts: () => ipcRenderer.invoke('serial:list'),
  openPort: (config) => ipcRenderer.invoke('serial:open', config),
  closePort: (path) => ipcRenderer.invoke('serial:close', path),
  writePort: (path, data) => ipcRenderer.invoke('serial:write', path, data),

  // 数据接收监听
  onSerialData: (callback) => {
    ipcRenderer.on('serial:data', (event, data) => callback(data));
  },

  // 文件操作
  saveFile: (path, data) => ipcRenderer.invoke('file:save', path, data),
  openFileDialog: () => ipcRenderer.invoke('file:open'),

  // 应用控制
  minimize: () => ipcRenderer.invoke('app:minimize'),
  maximize: () => ipcRenderer.invoke('app:maximize'),
  close: () => ipcRenderer.invoke('app:close')
});
```

## 4. 数据结构

### 4.1 串口配置

```typescript
interface SerialConfig {
  path: string;          // 串口路径 (COM1, /dev/ttyUSB0)
  baudRate: number;      // 波特率 (9600, 115200, ...)
  dataBits: 5 | 6 | 7 | 8;  // 数据位
  stopBits: 1 | 1.5 | 2;    // 停止位
  parity: 'none' | 'even' | 'odd' | 'mark' | 'space';  // 校验位
  flowControl: 'none' | 'hardware' | 'software';  // 流控制
}
```

### 4.2 数据包

```typescript
interface DataPacket {
  timestamp: number;     // 时间戳 (ms)
  direction: 'rx' | 'tx';  // 方向

  port: string;          // 来源串口
}
```

### 6.2 数据缓冲

```javascript
// 批量更新 UI，避免频繁重绘
class DataBuffer {
  constructor(flushInterval = 50) {
    this.buffer = [];
    this.flushInterval = flushInterval;

    setInterval(() => this.flush(), flushInterval);
  }

  push(data) {
    this.buffer.push(data);
  }

  flush() {
    if (this.buffer.length > 0) {
      this.emit('data', this.buffer);
      this.buffer = [];
    }
  }
}
```

### 7.2 协议解析器

```javascript
// 可扩展的协议解析器
class ProtocolParser {
  static parsers = new Map();

  static register(name, parser) {
    this.parsers.set(name, parser);
  }

  static parse(name, data) {
    const parser = this.parsers.get(name);
    return parser ? parser.parse(data) : data;
  }
}

// 注册 Modbus 解析器
```
